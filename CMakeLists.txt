cmake_minimum_required(VERSION 3.20)
# Keep assertions in the optimized validation build, as in MMModern.
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O2 -g" CACHE STRING "RelWithDebInfo flags")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g" CACHE STRING "RelWithDebInfo flags")
project(PSModern LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PSMODERN_ROM_TESTS "Register tests requiring external ROMs" ON)
set(PSMODERN_ROM_US "" CACHE FILEPATH "US Rev-A ROM for tests")
set(PSMODERN_ROM_BR "" CACHE FILEPATH "Brazilian ROM for tests")
set(PSMODERN_REFERENCE_DIR "" CACHE PATH "External reference emulator output")
set(PSMODERN_Z80_TEST_DIR "" CACHE PATH "External Z80 conformance program directory")

find_package(SDL2 CONFIG REQUIRED)
add_library(redcode_z80 STATIC third_party/redcode-z80/sources/Z80.c)
target_include_directories(redcode_z80 PUBLIC third_party/redcode-z80/API third_party/zeta/API)
target_compile_definitions(redcode_z80 PUBLIC Z80_STATIC)
target_compile_definitions(redcode_z80 PRIVATE Z80_WITH_Q Z80_WITH_ZILOG_NMOS_LD_A_IR_BUG)
add_library(psmodern_hw src/hw/cpu.cpp src/hw/memory.cpp src/machine/machine.cpp)
target_include_directories(psmodern_hw PUBLIC src)
target_link_libraries(psmodern_hw PUBLIC redcode_z80)
add_library(psmodern_rom src/rom/rom.cpp src/rom/sha1.cpp)
target_include_directories(psmodern_rom PUBLIC src)
add_executable(psmodern src/app/main.cpp)
target_link_libraries(psmodern PRIVATE psmodern_rom SDL2::SDL2)
if(MINGW)
  target_compile_options(psmodern_rom PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(psmodern PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(psmodern_z80_conformance tests/z80_conformance.cpp)
  target_link_libraries(psmodern_z80_conformance PRIVATE psmodern_hw)
  if(PSMODERN_Z80_TEST_DIR)
    set(_z80_test_prefix "${PSMODERN_Z80_TEST_DIR}/")
  else()
    set(_z80_test_prefix "")
  endif()
  foreach(suite zexdoc zexall)
    if(_z80_test_prefix)
      set(_z80_test_file "${_z80_test_prefix}${suite}.com")
    else()
      set(_z80_test_file "")
    endif()
    add_test(NAME ${suite} COMMAND psmodern_z80_conformance "${_z80_test_file}" cpm)
    set_tests_properties(${suite} PROPERTIES SKIP_RETURN_CODE 77 LABELS conformance)
  endforeach()
  foreach(suite z80flags z80memptr z80ccf)
    if(_z80_test_prefix)
      set(_z80_test_file "${_z80_test_prefix}${suite}.tap")
    else()
      set(_z80_test_file "")
    endif()
    add_test(NAME ${suite} COMMAND psmodern_z80_conformance "${_z80_test_file}" rak)
    set_tests_properties(${suite} PROPERTIES SKIP_RETURN_CODE 77 LABELS conformance)
  endforeach()
  if(_z80_test_prefix)
    set(_woodmass_file "${_z80_test_prefix}woodmass.tap")
    set(_spectrum_file "${_z80_test_prefix}spectrum48.rom")
  else()
    set(_woodmass_file "")
    set(_spectrum_file "")
  endif()
  foreach(suite wood-flags wood-memptr)
    add_test(NAME ${suite} COMMAND psmodern_z80_conformance "${_woodmass_file}" ${suite} "${_spectrum_file}")
    set_tests_properties(${suite} PROPERTIES SKIP_RETURN_CODE 77 LABELS conformance)
  endforeach()
  add_executable(psmodern_machine_tests tests/machine_tests.cpp)
  target_link_libraries(psmodern_machine_tests PRIVATE psmodern_hw)
  add_test(NAME machine_unit COMMAND psmodern_machine_tests)
  set_tests_properties(machine_unit PROPERTIES LABELS fast)
  add_executable(psmodern_memory_tests tests/memory_tests.cpp)
  target_link_libraries(psmodern_memory_tests PRIVATE psmodern_hw)
  add_test(NAME memory_unit COMMAND psmodern_memory_tests)
  set_tests_properties(memory_unit PROPERTIES LABELS fast)
  add_executable(psmodern_unit_tests tests/rom_tests.cpp)
  target_link_libraries(psmodern_unit_tests PRIVATE psmodern_rom)
  add_test(NAME rom_unit COMMAND psmodern_unit_tests "${CMAKE_CURRENT_BINARY_DIR}/rom-unit")
  set_tests_properties(rom_unit PROPERTIES LABELS fast)

  # Each ROM test executable receives a path and returns 77 when it is absent.
  function(psmodern_rom_test name path version)
    if(PSMODERN_ROM_TESTS)
      add_test(NAME "${name}" COMMAND psmodern_rom_tests "${path}" "${version}")
      set_tests_properties("${name}" PROPERTIES SKIP_RETURN_CODE 77 LABELS fast)
    endif()
  endfunction()
  if(PSMODERN_ROM_TESTS)
    add_executable(psmodern_rom_tests tests/real_rom_tests.cpp)
    target_link_libraries(psmodern_rom_tests PRIVATE psmodern_rom)
  endif()
  psmodern_rom_test(rom_us "${PSMODERN_ROM_US}" us)
  psmodern_rom_test(rom_br "${PSMODERN_ROM_BR}" br)
endif()
