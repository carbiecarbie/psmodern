  target_link_libraries(psmodern_unit_tests PRIVATE psmodern_rom)
  add_test(NAME rom_unit COMMAND psmodern_unit_tests "${CMAKE_CURRENT_BINARY_DIR}/rom-unit")
  set_tests_properties(rom_unit PROPERTIES LABELS fast)
  if(WIN32)
    add_test(NAME rom_unit_cleanup COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/rom_cleanup_tests.py $<TARGET_FILE:psmodern_unit_tests>)
    # Runs alone under ctest -j: its lock is released 0.1 s before the child's
    # ~0.2 s cleanup retries end, a margin a loaded machine can eat.
    set_tests_properties(rom_unit_cleanup PROPERTIES LABELS fast SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
  endif()

  # Each ROM test executable receives a path and returns 77 when it is absent.
  # Every test that needs a game ROM carries the label rom beside its own.
  function(psmodern_rom_test name path version)
    if(PSMODERN_ROM_TESTS)
      add_test(NAME "${name}_text" COMMAND psmodern_text_tests "${path}" "${version}")
      set_tests_properties("${name}_text" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      if(version STREQUAL "br")
        add_test(NAME "${name}_text_dialogue" COMMAND psmodern_text_recording_tests "${path}" "${version}"
          "${CMAKE_CURRENT_SOURCE_DIR}/recordings/br/text-dialogue.psrec")
        set_tests_properties("${name}_text_dialogue" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom)
      endif()
      add_test(NAME "${name}_controller" COMMAND psmodern_game_routine_tests "${path}" "${version}")
      set_tests_properties("${name}_controller" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      foreach(system random encounter experience money)
        add_test(NAME "${name}_${system}" COMMAND psmodern_${system}_tests "${path}" "${version}")
        set_tests_properties("${name}_${system}" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      endforeach()
      set(display_cart "")
      if(PSMODERN_REFERENCE_DIR)
        set(display_cart "${PSMODERN_REFERENCE_DIR}/${version}/deep-dungeon/cart-ram.bin")
      endif()
      add_test(NAME "${name}_experience_display" COMMAND psmodern_experience_display "${path}" "${version}"
        "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}/deep-dungeon.psrec" "${display_cart}")
      set_tests_properties("${name}_experience_display" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      add_test(NAME "${name}_money_display" COMMAND psmodern_money_display "${path}" "${version}"
        "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}/deep-dungeon.psrec" "${display_cart}")
      set_tests_properties("${name}_money_display" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      add_test(NAME "${name}_record" COMMAND psmodern_record_session "${path}" "${CMAKE_CURRENT_BINARY_DIR}/record-${version}" "${version}")
      set_tests_properties("${name}_record" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      add_test(NAME "${name}_replay" COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_rom_tests.py $<TARGET_FILE:psmodern_replay_runner> "${path}" "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}/short.psrec")
      set_tests_properties("${name}_replay" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      # Milestone 7 added three US dungeon recordings (a walk in two parts and a
      # Continue that leaves the dungeon); the Brazilian checkpoints stay six.
      set(checkpoints opening first-battle first-dungeon later-town boss deep-dungeon)
      if(version STREQUAL "us")
        list(APPEND checkpoints dungeon-walk-1 dungeon-walk-2 dungeon-continue)
      endif()
      foreach(checkpoint IN LISTS checkpoints)
        set(manifest "")
        if(PSMODERN_REFERENCE_DIR)
          set(manifest "${PSMODERN_REFERENCE_DIR}/${version}/${checkpoint}/manifest.txt")
        endif()
        add_test(NAME "${name}_${checkpoint}" COMMAND psmodern_checkpoint_tests
          "${path}" "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}/${checkpoint}.psrec" "${manifest}"
          "${CMAKE_CURRENT_BINARY_DIR}/routine-census/${version}/${checkpoint}.tsv")
        set_tests_properties("${name}_${checkpoint}" PROPERTIES SKIP_RETURN_CODE 77 LABELS "checkpoint;rom")
        # Both flags at 100 play the original; with the options on the play diverges and only the census is read.
        set(option_image "")
        if(PSMODERN_REFERENCE_DIR)
          set(option_image "${PSMODERN_REFERENCE_DIR}/${version}/${checkpoint}")
        endif()
        add_test(NAME "${name}_${checkpoint}_options" COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_rom_tests.py
          $<TARGET_FILE:psmodern_replay_runner> "${path}" "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}/${checkpoint}.psrec" options "${option_image}" full
          "${CMAKE_CURRENT_BINARY_DIR}/option-fights/${version}/${checkpoint}.txt")
        # Each test runs its five replays side by side, so ctest -j counts it as five processors.
        set_tests_properties("${name}_${checkpoint}_options" PROPERTIES SKIP_RETURN_CODE 77 LABELS "checkpoint;rom" FIXTURES_SETUP "options_${version}"
          PROCESSORS 5)
      endforeach()
      # At least one recording of the version starts a random fight in original mode, so that the rate-0
      # check of the options tests is known to be able to fail.
      add_test(NAME "${name}_options_fights" COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_rom_tests.py
        fights "${CMAKE_CURRENT_BINARY_DIR}/option-fights/${version}")
      set_tests_properties("${name}_options_fights" PROPERTIES SKIP_RETURN_CODE 77 LABELS "checkpoint;rom" FIXTURES_REQUIRED "options_${version}")
      # Milestone 7: cell-reader probes, table facts (compared with the other
      # version's tables when its ROM is configured), the movement and object
      # probes, the automap view and the automap session (its own work folder).
      if(version STREQUAL "us")
        set(other_rom "${PSMODERN_ROM_BR}")
      else()
        set(other_rom "${PSMODERN_ROM_US}")
      endif()
      add_test(NAME "${name}_dungeon_cells" COMMAND psmodern_dungeon_cells_tests "${path}" "${version}" cells)
      add_test(NAME "${name}_dungeon_tables" COMMAND psmodern_dungeon_cells_tests "${path}" "${version}" tables "${other_rom}")
      add_test(NAME "${name}_dungeon_movement" COMMAND psmodern_dungeon_movement_tests "${path}" "${version}")
      add_test(NAME "${name}_dungeon_objects" COMMAND psmodern_dungeon_objects_tests "${path}" "${version}")
      # The view test runs in three parts: the table checks and two halves of
      # the mutation of the original view drawing (states 0-199 and 200-399).
      add_test(NAME "${name}_automap_view" COMMAND psmodern_automap_view_tests "${path}" "${version}" tables)
      add_test(NAME "${name}_automap_mutation_0" COMMAND psmodern_automap_view_tests "${path}" "${version}" mutation-0)
      add_test(NAME "${name}_automap_mutation_1" COMMAND psmodern_automap_view_tests "${path}" "${version}" mutation-1)
      set_tests_properties("${name}_dungeon_cells" "${name}_dungeon_tables" "${name}_dungeon_movement" "${name}_dungeon_objects"
        "${name}_automap_view" "${name}_automap_mutation_0" "${name}_automap_mutation_1" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      # Each part has its own subfolder of the work folder. The session runs two
      # drivers on two threads and each passivity test four on four, so ctest -j
      # counts them as two and four processors.
      set(automap_session psmodern_automap_session "${path}" "${version}" "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}"
        "${PSMODERN_REFERENCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/automap-session-${version}")
      add_test(NAME "${name}_automap_session" COMMAND ${automap_session} session)
      set_tests_properties("${name}_automap_session" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom PROCESSORS 2)
      # Brazilian boss (19,550 frames) is left out: alone it takes about 45 s.
      set(passivity_recordings first-dungeon deep-dungeon)
      if(version STREQUAL "us")
        list(APPEND passivity_recordings boss dungeon-walk-1 dungeon-walk-2 dungeon-continue)
        # The wrong-map paths: a new game and Continues after exploring, in
        # one driver, from the walk recordings.
        add_test(NAME "${name}_automap_wrong_map" COMMAND ${automap_session} wrong-map)
        set_tests_properties("${name}_automap_wrong_map" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom)
        # The walk chain plays the three walk recordings in one save folder,
        # one link each, in order: each Continue finds the previous save's map.
        foreach(link 1 2 3)
          add_test(NAME "${name}_automap_chain_${link}" COMMAND ${automap_session} chain ${link})
          set_tests_properties("${name}_automap_chain_${link}" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom FIXTURES_SETUP "automap_chain_${link}")
          if(link GREATER 1)
            math(EXPR previous "${link} - 1")
            set_tests_properties("${name}_automap_chain_${link}" PROPERTIES FIXTURES_REQUIRED "automap_chain_${previous}")
          endif()
        endforeach()
      endif()
      foreach(recording IN LISTS passivity_recordings)
        add_test(NAME "${name}_automap_passivity_${recording}" COMMAND ${automap_session} passivity ${recording})
        set_tests_properties("${name}_automap_passivity_${recording}" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom PROCESSORS 4)
      endforeach()
      # Milestone 10: the map reader and state probes, the driver probes and
      # every recording with the widescreen option off and on. Each recording
      # test runs two drivers on two threads, so ctest -j counts it as two.
      add_test(NAME "${name}_wide_view_reader" COMMAND psmodern_wide_view_tests rom "${path}" "${version}")
      set_tests_properties("${name}_wide_view_reader" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      add_test(NAME "${name}_wide_view_probes" COMMAND psmodern_wide_view_tests probes "${path}" "${version}"
        "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}" "${PSMODERN_REFERENCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/wide-view-${version}/probes" "${other_rom}")
      set_tests_properties("${name}_wide_view_probes" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom)
      # The space flight from later-town's start image: routes 81h-84h with the
      # option switched on mid-flight and a skipped flight, then 85h-88h.
      add_test(NAME "${name}_wide_view_ship" COMMAND psmodern_wide_view_tests ship "${path}" "${version}"
        "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}" "${PSMODERN_REFERENCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/wide-view-${version}/ship" "81,82,83,84,extra")
      add_test(NAME "${name}_wide_view_ship_routes" COMMAND psmodern_wide_view_tests ship "${path}" "${version}"
        "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}" "${PSMODERN_REFERENCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/wide-view-${version}/ship-routes" "85,86,87,88")
      set_tests_properties("${name}_wide_view_ship" "${name}_wide_view_ship_routes" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom)
      set(wide_recordings opening first-battle first-dungeon later-town boss deep-dungeon short)
      if(version STREQUAL "us")
        list(APPEND wide_recordings dungeon-walk-1 dungeon-walk-2 dungeon-continue)
      else()
        list(APPEND wide_recordings text-dialogue)
      endif()
      foreach(recording IN LISTS wide_recordings)
        add_test(NAME "${name}_wide_view_${recording}" COMMAND psmodern_wide_view_tests recording "${path}" "${version}"
          "${CMAKE_CURRENT_SOURCE_DIR}/recordings/${version}" "${PSMODERN_REFERENCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/wide-view-${version}/${recording}" "${recording}")
        set_tests_properties("${name}_wide_view_${recording}" PROPERTIES SKIP_RETURN_CODE 77 LABELS rom PROCESSORS 2)
      endforeach()
      add_test(NAME "${name}" COMMAND psmodern_rom_tests "${path}" "${version}")
      set_tests_properties("${name}" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      if(version STREQUAL "us")
      add_test(NAME "${name}_save_process" COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/save_process_tests.py $<TARGET_FILE:psmodern> "${path}")
      # Runs alone under ctest -j: it gives real game processes 1 to 5 s of wall-clock time.
      set_tests_properties("${name}_save_process" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom" RUN_SERIAL TRUE)
      endif()
      add_test(NAME "${name}_app" COMMAND psmodern_app_rom_tests "${path}" "${version}")
      set_tests_properties("${name}_app" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      add_test(NAME "${name}_start_screen" COMMAND psmodern_launcher_tests "${path}" "${version}")
      set_tests_properties("${name}_start_screen" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom")
      if(version STREQUAL "us")
      add_test(NAME "${name}_unicode_process" COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/unicode_process_tests.py $<TARGET_FILE:psmodern> "${path}")
      # Runs alone under ctest -j: it gives a real game process 1 s of wall-clock time.
      set_tests_properties("${name}_unicode_process" PROPERTIES SKIP_RETURN_CODE 77 LABELS "fast;rom" RUN_SERIAL TRUE)
      endif()
    endif()
  endfunction()
  if(PSMODERN_ROM_TESTS)
    add_executable(psmodern_record_session tests/record_session.cpp src/app/sdl_host.cpp)

    add_executable(psmodern_rom_tests tests/real_rom_tests.cpp)

    add_executable(psmodern_app_rom_tests tests/app_rom_tests.cpp)
    target_link_libraries(psmodern_app_rom_tests PRIVATE psmodern_app)
  endif()

  psmodern_rom_test(rom_br "${PSMODERN_ROM_BR}" br)
endif()
