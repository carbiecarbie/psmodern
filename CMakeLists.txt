  # Milestone 14: the shipped program must contain neither the Z80 core nor the
  # test oracle. The link closure of psmodern is computed here; the script
  # also inspects the symbols of the program and of the game libraries.
  function(psmodern_link_closure result)
    set(_pending ${ARGN})
    set(_seen)
    while(_pending)
      list(POP_FRONT _pending _item)
      if(_item IN_LIST _seen OR NOT TARGET ${_item})
        continue()
      endif()
      list(APPEND _seen ${_item})
      get_target_property(_type ${_item} TYPE)
      if(_type STREQUAL "INTERFACE_LIBRARY")
        set(_props INTERFACE_LINK_LIBRARIES)
      else()
        set(_props LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
      endif()
      foreach(_prop ${_props})
        get_target_property(_deps ${_item} ${_prop})
        if(_deps)
          list(APPEND _pending ${_deps})
        endif()
      endforeach()
    endwhile()
    set(${result} "${_seen}" PARENT_SCOPE)
  endfunction()
  psmodern_link_closure(_program_closure psmodern)
  string(REPLACE ";" "," _program_closure_text "${_program_closure}")
  # Review R3 L7: every static library of the program's own targets in the closure is scanned.
  set(_program_archives)
  foreach(_target IN LISTS _program_closure)
    get_target_property(_type ${_target} TYPE)
    get_target_property(_imported ${_target} IMPORTED)
    if(_type STREQUAL "STATIC_LIBRARY" AND NOT _imported)
      list(APPEND _program_archives "$<TARGET_FILE:${_target}>")
    endif()
  endforeach()
